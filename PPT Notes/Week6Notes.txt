namespace Week6Learnings
{
    public class Person
    {
        public string Name { get; set; }
        // In a graph, "Children" becomes "Friends" or "Neighbors"
        public List<Person> Friends { get; set; } = new List<Person>();

        public Person(string name) => Name = name;

        public void AddFriend(Person friend)
        {
            if (!Friends.Contains(friend))
            {
                Friends.Add(friend);
                // In a social network, friendship is usually two-way (Undirected)
                friend.Friends.Add(this);
            }
        }
    }

    public class SocialNetworkGraph
    {
        private List<Person> _members = new List<Person>();

        public void AddMember(Person person) => _members.Add(person);

        public void ShowNetwork()
        {
            Console.WriteLine("Social Network Map:");
            foreach (var person in _members)
            {       
                var friendNames = new List<string>();
                foreach (var friend in person.Friends)
                {
                    friendNames.Add(friend.Name);
                }

                Console.WriteLine($"{person.Name} is friends with: {string.Join(", ", friendNames)}");
            }
        }
    }
    internal class Demo07GraphFriends
    {
        static void Main()
        {
            var graph = new SocialNetworkGraph();

            // Create People
            var aman = new Person("Aman");
            var bhaskar = new Person("Bhaskar");
            var chetan = new Person("Chetan");
            var dinkar = new Person("Dinkar");

            graph.AddMember(aman);
            graph.AddMember(bhaskar);
            graph.AddMember(chetan);
            graph.AddMember(dinkar);

            // Create Connections (Edges)
            aman.AddFriend(bhaskar);     // Alice <-> Bob
            bhaskar.AddFriend(chetan);   // Bob <-> Charlie
            chetan.AddFriend(aman); // Charlie <-> Alice (This forms a CYCLE - impossible in a Tree!)
            chetan.AddFriend(dinkar); // Charlie <-> David

            graph.ShowNetwork();
        }
    }
}
