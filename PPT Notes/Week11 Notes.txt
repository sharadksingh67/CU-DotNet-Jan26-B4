public interface ILifecycleService
{
    Guid GetId();
}

public class LifecycleService : ILifecycleService
{
    private readonly Guid _id;
    public LifecycleService() => _id = Guid.NewGuid();
    public Guid GetId() => _id;
}


// Each registration gets a different lifetime
builder.Services.AddTransient<ITransientService, LifecycleService>();
builder.Services.AddScoped<IScopedService, LifecycleService>();
builder.Services.AddSingleton<ISingletonService, LifecycleService>();


[Route("api/[controller]")]
public class LifetimeController : ControllerBase
{
    private readonly ITransientService _transient1;
    private readonly ITransientService _transient2;
    private readonly IScopedService _scoped1;
    private readonly IScopedService _scoped2;
    private readonly ISingletonService _singleton;

    public LifetimeController(
        ITransientService t1, ITransientService t2,
        IScopedService s1, IScopedService s2,
        ISingletonService singleton)
    {
        _transient1 = t1; _transient2 = t2;
        _scoped1 = s1; _scoped2 = s2;
        _singleton = singleton;
    }

    [HttpGet]
    public IActionResult Get()
    {
        return Ok(new {
            Transient = new { ID1 = _transient1.GetId(), ID2 = _transient2.GetId() },
            Scoped = new { ID1 = _scoped1.GetId(), ID2 = _scoped2.GetId() },
            Singleton = _singleton.GetId()
        });
    }
}
