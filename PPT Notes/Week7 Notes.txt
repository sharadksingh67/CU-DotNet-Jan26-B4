using System;
using Microsoft.Data.SqlClient; // Or System.Data.SqlClient
using System.Data;

public void GetCategoryInventory(int categoryId)
{
    string connectionString = "Your_Connection_String_Here";

    using (SqlConnection conn = new SqlConnection(connectionString))
    {
        // 1. Create the Command
        SqlCommand cmd = new SqlCommand("proc_GetCatInventory", conn);
        cmd.CommandType = CommandType.StoredProcedure;

        // 2. Add Input Parameter
        cmd.Parameters.AddWithValue("@catid", categoryId);

        // 3. Add Output Parameter
        SqlParameter outputParam = new SqlParameter
        {
            ParameterName = "@totalInv",
            SqlDbType = SqlDbType.Decimal,
            Precision = 10,
            Scale = 2,
            Direction = ParameterDirection.Output
        };
        cmd.Parameters.Add(outputParam);

        try
        {
            conn.Open();
            
            // 4. Execute the command 
            // We use ExecuteNonQuery because the proc doesn't return a result set (SELECT), 
            // it only assigns a value to a variable.
            cmd.ExecuteNonQuery();

            // 5. Retrieve the Output Value
            decimal totalInventory = (decimal)outputParam.Value;

            Console.WriteLine($"Total Inventory for Category {categoryId}: {totalInventory:C}");
        }
        catch (Exception ex)
        {
            Console.WriteLine("Error: " + ex.Message);
        }
    }
}
